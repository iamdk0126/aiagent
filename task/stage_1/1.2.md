# 階段 1.2 設計規格書：模型商城資料結構

## 1. 階段目標與職責邊界
本階段的核心目標是建立模型商城的資料庫層（Repositories）與 ORM 操作介面，確保模型商城功能的資料持久化與業務邏輯分離。階段產出將包含：
- `database/repositories/model_store_repo.py`：模型商城的資料庫操作層
- `schemas/asset_schemas.py`：模型商城的 Pydantic 資料模型

此階段將遵循以下設計原則：
1. 所有業務邏輯應在 Service 層實現，資料庫操作層僅負責資料存取
2. 使用 SQLAlchemy ORM 進行資料庫操作，避免原始 SQL
3. 資料模型應與註冊表中的資料表結構保持一致
4. 提供清晰的 CRUD 方法簽章，確保可測試性

class ModelStore(Base):
    __tablename__ = 'model_store'
    id = Column(Integer, primary_key=True)
    model_id = Column(String, unique=True, nullable=False)
    model_name = Column(String, nullable=False)
    provider_id = Column(String)
    is_local = Column(Integer)
    is_commercial = Column(Integer)
    model_path = Column(String, nullable=False)
    version = Column(String, nullable=False)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class ModelStoreRepository:
    def __init__(self, db_path: str):
        self.engine = create_engine(f'sqlite:///{db_path}')
        self.Session = scoped_session(sessionmaker(bind=self.engine, autoflush=False, autocommit=False))
        self.session = self.Session()
        
    def create_all(self) -> None:
        Base.metadata.create_all(self.engine)
        
    def init_db(self) -> None:
        if not os.path.exists(db_path):
            self.session.close()
            self.Session.remove()
            Base.metadata.create_all(self.engine)
            self.Session.configure(bind=self.engine)
            self.session = self.Session()
            
    def get_model_by_id(self, model_id: str) -> Optional[ModelStore]:
        return self.session.get(ModelStore, model_id)
        
    def get_model_by_name(self, model_name: str) -> Optional[ModelStore]:
        return self.session.query(ModelStore).filter(ModelStore.model_name == model_name).first()
        
    def get_all_models(self) -> List[ModelStore]:
        return self.session.query(ModelStore).all()
        
    def add_model(self, model_id: str, model_name: str, provider_id: str, is_local: int, is_commercial: int, model_path: str, version: str) -> None:
        new_model = ModelStore(
            model_id=model_id,
            model_name=model_name,
            provider_id=provider_id,
            is_local=is_local,
            is_commercial=is_commercial,
            model_path=model_path,
            version=version
        )
        self.session.add(new_model)
        self.session.commit()
        
    def update_model(self, model_id: str, model_name: str, provider_id: str, is_local: int, is_commercial: int, model_path: str, version: str) -> None:
        self.session.query(ModelStore).filter(ModelStore.model_id == model_id).update({

            'is_local': is_local,
            'is_commercial': is_commercial,
            'model_path': model_path,

    def delete_model(self, model_id: str) -> None:
        self.session.query(ModelStore).filter(ModelStore.model_id == model_id).delete()
        self.session.commit()

from typing import List, Optional, Dict, Any
from datetime import datetime

class ModelStore(BaseModel):
    model_id: str = Field(..., description="模型唯一識別符")
    model_name: str = Field(..., description="模型名稱")
    provider_id: str = Field(..., description="提供者 ID")

    version: str = Field(..., description="模型版本")
    created_at: datetime = Field(..., description="創建時間")
    updated_at: datetime = Field(..., description="更新時間")

class ModelStoreManager:
    def __init__(self, db_path: str):
        self.repo = ModelStoreRepository(db_path)
        
    def get_all_models(self) -> List[ModelStore]:
        return [ModelStore(**model) for model in self.repo.get_all_models()]
        
    def get_model_by_id(self, model_id: str) -> Optional[ModelStore]:
        return self.repo.get_model_by_id(model_id)
